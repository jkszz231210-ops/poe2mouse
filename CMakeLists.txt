cmake_minimum_required(VERSION 3.22)
project(poe2mouse VERSION 0.1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(input_core INTERFACE)
target_include_directories(input_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
function(strict target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE input_core)
  strict(core_tests)
  add_test(NAME core_tests COMMAND core_tests)
endif()
if(WIN32)
  add_executable(input_lab WIN32 src/windows/input_lab.cpp)
  target_link_libraries(input_lab PRIVATE input_core user32 gdi32 comdlg32 xinput9_1_0)
  target_compile_definitions(input_lab PRIVATE UNICODE _UNICODE NOMINMAX)
  strict(input_lab)
  add_custom_command(TARGET input_lab POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/profiles/lab.profile" "$<TARGET_FILE_DIR:input_lab>/lab.default.profile")
  install(TARGETS input_lab RUNTIME DESTINATION .)
  install(FILES profiles/lab.profile DESTINATION . RENAME lab.default.profile)
  install(FILES docs/HOME_TEST.md docs/FEEDBACK.md docs/BINDINGS.md DESTINATION .)
endif()
