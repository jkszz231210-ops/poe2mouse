cmake_minimum_required(VERSION 3.22)
project(poe2mouse VERSION 0.1.16 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(input_core INTERFACE)
target_include_directories(input_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
function(strict target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE input_core)
  strict(core_tests)
  add_test(NAME core_tests COMMAND core_tests)
endif()
if(WIN32)
  add_executable(input_lab WIN32 src/windows/input_lab.cpp)
  target_link_libraries(input_lab PRIVATE input_core user32 gdi32 comdlg32 xinput9_1_0)
  target_compile_definitions(input_lab PRIVATE UNICODE _UNICODE NOMINMAX)
  strict(input_lab)
  add_custom_command(TARGET input_lab POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/profiles/lab.profile" "$<TARGET_FILE_DIR:input_lab>/lab.default.profile")
  add_executable(device_probe src/windows/device_probe.cpp)
  target_link_libraries(device_probe PRIVATE input_core xinput9_1_0)
  target_compile_definitions(device_probe PRIVATE UNICODE _UNICODE NOMINMAX)
  strict(device_probe)
  if(MINGW)
    target_link_options(device_probe PRIVATE -municode)
  endif()
  add_executable(input_bridge src/windows/input_bridge.cpp)
  target_link_libraries(input_bridge PRIVATE input_core user32 gdi32 comdlg32 shell32 xinput9_1_0)
  target_compile_definitions(input_bridge PRIVATE UNICODE _UNICODE NOMINMAX)
  strict(input_bridge)
  if(MINGW)
    target_link_options(input_bridge PRIVATE -municode)
  endif()
  install(TARGETS input_lab device_probe input_bridge RUNTIME DESTINATION .)
  install(FILES scripts/PrepareDependencies.ps1 DESTINATION .)
  install(FILES profiles/bridge.default.profile profiles/bridge.default.map profiles/bridge.default.options DESTINATION .)
  install(FILES profiles/lab.profile DESTINATION . RENAME lab.default.profile)
  install(FILES docs/HOME_TEST.md docs/FEEDBACK.md docs/BINDINGS.md docs/DEVICE_PROBE.md docs/REFERENCE_STUDY.md docs/BRIDGE_TEST.md docs/ACCEPTANCE.md docs/ISSUE_LOG.md docs/QUICK_START.md docs/NATIVE_UI.md DESTINATION .)
endif()
